cmake_minimum_required(VERSION 3.20)
project(TrafficLimitService LANGUAGES CXX)
set(WINDIVERT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/WinDivert")
if(NOT EXISTS "${WINDIVERT_ROOT}/include/windivert.h" OR NOT EXISTS "${WINDIVERT_ROOT}/x64/WinDivert.lib")
  message(FATAL_ERROR "Bundled WinDivert x64 SDK is required")
endif()
set(CMAKE_CXX_STANDARD 20)
add_executable(TrafficLimitService WIN32 main.cpp service.cpp traffic_counter.cpp limiter.cpp process_stats.cpp)
target_compile_definitions(TrafficLimitService PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601 NTDDI_VERSION=0x06010000 TRAFFIC_LIMIT_USE_WINDIVERT)
target_include_directories(TrafficLimitService PRIVATE "${WINDIVERT_ROOT}/include")
target_link_directories(TrafficLimitService PRIVATE "${WINDIVERT_ROOT}/x64")
target_link_libraries(TrafficLimitService PRIVATE iphlpapi advapi32 userenv ws2_32 WinDivert)
add_custom_command(TARGET TrafficLimitService POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINDIVERT_ROOT}/x64/WinDivert.dll" "$<TARGET_FILE_DIR:TrafficLimitService>/WinDivert.dll"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINDIVERT_ROOT}/x64/WinDivert64.sys" "$<TARGET_FILE_DIR:TrafficLimitService>/WinDivert64.sys")
install(TARGETS TrafficLimitService RUNTIME DESTINATION .)
