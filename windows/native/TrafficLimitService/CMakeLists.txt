cmake_minimum_required(VERSION 3.20)
project(TrafficLimitService LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(TrafficLimitService main.cpp service.cpp traffic_counter.cpp limiter.cpp)
target_compile_definitions(TrafficLimitService PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601 NTDDI_VERSION=0x06010000)
target_link_libraries(TrafficLimitService PRIVATE iphlpapi advapi32 userenv ws2_32)
option(TRAFFIC_LIMIT_USE_WINDIVERT "Build packet limiter using the signed WinDivert driver" OFF)
if(TRAFFIC_LIMIT_USE_WINDIVERT)
 target_compile_definitions(TrafficLimitService PRIVATE TRAFFIC_LIMIT_USE_WINDIVERT)
 target_include_directories(TrafficLimitService PRIVATE third_party/WinDivert/include)
 target_link_directories(TrafficLimitService PRIVATE third_party/WinDivert/x64)
 target_link_libraries(TrafficLimitService PRIVATE WinDivert)
endif()
install(TARGETS TrafficLimitService RUNTIME DESTINATION .)
